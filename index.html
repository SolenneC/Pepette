// Codes secrets simples const PASSCODES = { 1: "EGEE01", 2: "HERA02", 3: "ARES03", 4: "HEST04", 5: "ZEUS05", 6: "APOL06", 7: "ARTE07", 8: "POSE08", 9: "DEME09", 10: "ATHE10", 11: "HEPH11", 12: "AFRO12" }; function verifyPasscode() { const input = document.getElementById('passcode-input').value.trim().toUpperCase(); if (input === PASSCODES[selectedChapter]) { localStorage.setItem(`chapter_${selectedChapter}_unlocked`, 'true'); showContent(selectedChapter); } else { document.getElementById('error-msg').style.display = 'block'; } }
